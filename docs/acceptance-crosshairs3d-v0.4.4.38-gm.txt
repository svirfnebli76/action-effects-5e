await (async () => {
  const TAG = "AE5E Checkpoint 3 Extended GM Acceptance 02";
  const VERSION = "0.4.4.38";
  const MODULE_ID = "action-effects-5e";
  const module = game.modules.get(MODULE_ID);
  const api = module?.api?.crosshairs3d;
  if (!game.user.isGM) return ui.notifications.warn(`${TAG} requires a GM.`);
  if (module?.version !== VERSION || !api?.propagation?.apply || !api?.persistent?.resolveAttached) {
    return ui.notifications.warn(`Install AE5E ${VERSION}, enable it, and reload first.`);
  }
  if (!canvas.ready || !canvas.grid.isSquare) return ui.notifications.warn("Open a square-grid Scene first.");
  const source = canvas.tokens.placeables.find(token => token.name === "Caerwyn Thorne");
  if (!source) return ui.notifications.warn("Caerwyn Thorne must be present on this Scene.");

  const scene = canvas.scene;
  const rect = canvas.dimensions.sceneRect;
  const size = scene.grid.size;
  const distance = scene.grid.distance;
  const metrics = Object.freeze({ size, distance, originX: rect.x, originY: rect.y,
    grid: Object.freeze({ distance, origin: Object.freeze({ x: 0, y: 0, z: 0 }) }) });
  const sourceSnapshot = { x: source.document.x, y: source.document.y,
    elevation: source.document.elevation, rotation: source.document.rotation };
  const targetSnapshot = [...game.user.targets].map(token => token.id);
  const temporary = { walls: [], levels: [], regions: [] };
  const checks = [];
  const record = (check, passed, details = {}) => checks.push({ check, passed: Boolean(passed), details });
  const keys = cells => new Set(cells.map(cell => `${cell.x},${cell.y},${cell.z}`));
  const same = (a, b) => a.size === b.size && [...a].every(key => b.has(key));
  const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
  async function waitFor(predicate, label, timeout = 5000) {
    const started = performance.now();
    while (performance.now() - started < timeout) {
      const value = await predicate();
      if (value) return value;
      await wait(50);
    }
    throw new Error(`Timed out waiting for ${label}.`);
  }
  const distancePoint = (x, y, z) => ({
    x: ((x - rect.x) / size) * distance,
    y: ((y - rect.y) / size) * distance,
    z
  });
  const toPixels = point => ({
    x: rect.x + (point.x / distance * size),
    y: rect.y + (point.y / distance * size)
  });

  try {
    await source.document.update({ x: 2500, y: 2500, elevation: 0, rotation: 0 }, { animate: false });
    const center = distancePoint(2550, 2550, 0);

    // A cancelled placement must restore targets and must never create its requested persistent Region.
    const beforeCancelRegions = scene.regions.size;
    const beforeCancelTargets = new Set([...game.user.targets].map(token => token.id));
    const controller = new AbortController();
    const placement = api.show({ source, shape: { type: "prism", width: 10, length: 10, height: 5 },
      range: { max: 60 }, signal: controller.signal, persistent: { enabled: true } });
    await waitFor(() => api.placement.getStats().active, "placement activation");
    controller.abort();
    const cancelled = await placement;
    const afterCancelTargets = new Set([...game.user.targets].map(token => token.id));
    record("Cancellation restores targets and creates no persistent Region",
      cancelled.cancelled && scene.regions.size === beforeCancelRegions
        && same(beforeCancelTargets, afterCancelTargets),
      { cancelled: cancelled.cancelled, beforeRegions: beforeCancelRegions, afterRegions: scene.regions.size });

    // Exact unobstructed prisms use one native rectangle rather than a cell proxy.
    const nativeShape = api.geometry.normalizeShape({ type: "prism",
      origin: { x: center.x + 10, y: center.y, z: 2.5 }, width: 10, length: 20, height: 7.5, yaw: 30 });
    const nativePropagation = await api.propagation.apply({ shape: nativeShape, grid: metrics.grid, mode: "none" });
    const nativeCreate = await api.persistent.create({ propagation: nativePropagation, scene, source, metrics,
      options: { name: `${TAG} Native`, locked: true } });
    temporary.regions.push(nativeCreate.regionUuid);
    const nativeRegion = await fromUuid(nativeCreate.regionUuid);
    const nativeMeta = nativeRegion?.getFlag(MODULE_ID, "crosshair3dPersistentArea");
    record("Exact None prism uses one native Foundry Region",
      nativeMeta?.backend === "native" && nativeRegion?.shapes?.length === 1
        && nativeRegion.shapes[0].type === "rectangle"
        && !nativeRegion.getFlag(MODULE_ID, "regionCells"),
      { backend: nativeMeta?.backend, type: nativeRegion?.shapes?.[0]?.type });

    // A Wall assigned only to 0–5 ft must not block a horizontal 10–15 ft propagation ray.
    const rayOrigin = { x: center.x, y: center.y, z: 10 };
    const highShape = api.geometry.normalizeShape({ type: "prism",
      origin: { x: center.x + 20, y: center.y, z: 10 }, width: 10, length: 10, height: 5, yaw: 0 });
    const highClear = await api.propagation.apply({ shape: highShape, grid: metrics.grid, mode: "direct",
      origin: rayOrigin, environment: api.propagation.createFoundryEnvironment({ scene, source, metrics }) });
    const levelData = { name: `${TAG} 0-5 ft`, elevation: { bottom: 0, top: 5 } };
    const [level] = await scene.createEmbeddedDocuments("Level", [levelData]);
    temporary.levels.push(level.id);
    const wallX = toPixels({ x: center.x + 10, y: center.y }).x;
    const wallA = toPixels({ x: center.x + 10, y: center.y - 15 });
    const wallB = toPixels({ x: center.x + 10, y: center.y + 15 });
    const normal = CONST.EDGE_SENSE_TYPES?.NORMAL ?? CONST.WALL_SENSE_TYPES?.NORMAL ?? 1;
    const [wall] = await scene.createEmbeddedDocuments("Wall", [{
      c: [wallX, wallA.y, wallX, wallB.y], move: normal, sight: 0, light: 0, sound: 0,
      door: 0, ds: 0, levels: [level.id]
    }]);
    temporary.walls.push(wall.id);
    const bounded = await api.propagation.apply({ shape: highShape, grid: metrics.grid, mode: "direct",
      origin: rayOrigin, environment: api.propagation.createFoundryEnvironment({ scene, source, metrics }) });
    await wall.update({ levels: [] });
    const unbounded = await api.propagation.apply({ shape: highShape, grid: metrics.grid, mode: "direct",
      origin: rayOrigin, environment: api.propagation.createFoundryEnvironment({ scene, source, metrics }) });
    record("Wall Level height is honored independently of the viewed Level",
      same(keys(highClear.cells), keys(bounded.cells)) && unbounded.cells.length < bounded.cells.length,
      { clear: highClear.cells.length, bounded: bounded.cells.length, unbounded: unbounded.cells.length });
    await scene.deleteEmbeddedDocuments("Wall", [wall.id]);
    temporary.walls = temporary.walls.filter(id => id !== wall.id);

    // A core Define Surface Region must block upward Direct rays without relying on Wall Levels.
    const risingOrigin = { x: center.x, y: center.y, z: 0 };
    const risingShape = api.geometry.normalizeShape({ type: "prism",
      origin: { x: center.x + 20, y: center.y, z: 15 }, width: 10, length: 10, height: 5, yaw: 0 });
    const risingClear = await api.propagation.apply({ shape: risingShape, grid: metrics.grid, mode: "direct",
      origin: risingOrigin, environment: api.propagation.createFoundryEnvironment({ scene, source, metrics }) });
    const surfaceTopLeft = toPixels({ x: center.x - 5, y: center.y - 20 });
    const [surface] = await scene.createEmbeddedDocuments("Region", [{
      name: `${TAG} Surface`, color: "#ff9900",
      shapes: [{ type: "rectangle", x: surfaceTopLeft.x, y: surfaceTopLeft.y,
        width: 40 / distance * size, height: 40 / distance * size,
        anchorX: 0, anchorY: 0, rotation: 0, gridBased: false }],
      elevation: { bottom: 10, top: 10 },
      behaviors: [{ type: "defineSurface", system: { placement: "bottom", move: true,
        light: false, sight: false, sound: false, occlusion: false, exposure: false, culling: false } }]
    }]);
    temporary.regions.push(surface.uuid);
    await wait(50);
    const surfaceBlocked = await api.propagation.apply({ shape: risingShape, grid: metrics.grid, mode: "direct",
      origin: risingOrigin, environment: api.propagation.createFoundryEnvironment({ scene, source, metrics }) });
    record("Foundry Define Surface blocks Direct propagation at its own elevation",
      surfaceBlocked.cells.length < risingClear.cells.length,
      { clear: risingClear.cells.length, surface: surfaceBlocked.cells.length });
    await scene.deleteEmbeddedDocuments("Region", [surface.id]);
    temporary.regions = temporary.regions.filter(uuid => uuid !== surface.uuid);

    // Attached Direct areas must be re-resolved after the source moves.
    const attachedShape = api.geometry.normalizeShape({ type: "prism",
      origin: { x: center.x + 10, y: center.y, z: 0 }, width: 5, length: 5, height: 5, yaw: 0 });
    const attachedPropagation = await api.propagation.apply({ shape: attachedShape, grid: metrics.grid,
      mode: "direct", origin: center,
      environment: api.propagation.createFoundryEnvironment({ scene, source, metrics }) });
    const attachedCreate = await api.persistent.create({ propagation: attachedPropagation, scene, source, metrics,
      options: { name: `${TAG} Attached Direct`, attached: true, locked: true } });
    temporary.regions.push(attachedCreate.regionUuid);
    const attachedBefore = await fromUuid(attachedCreate.regionUuid);
    const beforeMeta = attachedBefore.getFlag(MODULE_ID, "crosshair3dPersistentArea");
    await source.document.update({ x: 2500 + size }, { animate: false });
    const attachedAfter = await waitFor(async () => {
      const region = await fromUuid(attachedCreate.regionUuid);
      const metadata = region?.getFlag(MODULE_ID, "crosshair3dPersistentArea");
      return metadata?.sourceTransform?.x === 2500 + size && metadata?.repropagatedAt
        ? { region, metadata } : null;
    }, "attached Direct re-propagation");
    const afterConfig = attachedAfter.region.getFlag(MODULE_ID, "regionCells");
    record("Attached Direct area re-resolves after source movement",
      attachedAfter.metadata.backend === "cells"
        && Math.abs(attachedAfter.metadata.shape.origin.x - beforeMeta.shape.origin.x - distance) < 1e-7
        && Object.keys(afterConfig?.cells ?? {}).length > 0,
      { beforeX: beforeMeta.shape.origin.x, afterX: attachedAfter.metadata.shape.origin.x,
        cells: Object.keys(afterConfig?.cells ?? {}).length,
        reason: attachedAfter.metadata.lastReason });

    const passed = checks.every(check => check.passed);
    console.group(TAG);
    console.table(checks.map(row => ({ check: row.check, passed: row.passed,
      details: JSON.stringify(row.details) })));
    console.log({ tag: TAG, version: VERSION, passed, checks });
    console.groupEnd();
    ui.notifications[passed ? "info" : "warn"](`${TAG}: ${passed ? "PASS" : "CHECK CONSOLE"}.`);
    return { tag: TAG, version: VERSION, passed, checks };
  } catch (error) {
    console.error(`${TAG}: FAIL`, error);
    ui.notifications.error(`${TAG}: ${error.message}`);
    return { tag: TAG, version: VERSION, passed: false, error: error.message, checks };
  } finally {
    for (const uuid of [...temporary.regions].reverse()) {
      try {
        const region = await fromUuid(uuid);
        if (!region) continue;
        if (module.api.regions.isOwned(region)) await module.api.regions.delete(uuid);
        else await scene.deleteEmbeddedDocuments("Region", [region.id]);
      } catch (error) { console.warn(`${TAG}: temporary Region cleanup failed`, uuid, error); }
    }
    try { if (temporary.walls.length) await scene.deleteEmbeddedDocuments("Wall", temporary.walls); }
    catch (error) { console.warn(`${TAG}: temporary Wall cleanup failed`, error); }
    try { if (temporary.levels.length) await scene.deleteEmbeddedDocuments("Level", temporary.levels); }
    catch (error) { console.warn(`${TAG}: temporary Level cleanup failed`, error); }
    try { await source.document.update(sourceSnapshot, { animate: false }); }
    catch (error) { console.warn(`${TAG}: source restoration failed`, error); }
    try { canvas.tokens.setTargets(targetSnapshot.filter(id => canvas.tokens.get(id)), { mode: "replace" }); }
    catch (_) {}
  }
})();
