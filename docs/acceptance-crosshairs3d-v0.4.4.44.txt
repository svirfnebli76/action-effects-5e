await (async () => {
  const TAG = "AE5E Checkpoint 4 Production Acceptance 01";
  const VERSION = "0.4.4.44";
  const MODULE_ID = "action-effects-5e";
  const module = game.modules.get(MODULE_ID);
  const api = module?.api?.crosshairs3d;

  if (!game.user.isGM) {
    return ui.notifications.warn(`${TAG} must be run by the primary GM.`);
  }
  if (module?.version !== VERSION || !api?.showConfigured || !api?.configuration?.resolve) {
    return ui.notifications.warn(`Install and activate AE5E ${VERSION}, then reload Foundry.`);
  }
  if (!canvas.ready || !canvas.grid.isSquare) {
    return ui.notifications.warn("Open a square-grid Scene first.");
  }
  if (api.getStats().active) {
    return ui.notifications.warn("Finish or cancel the current 3D placement first.");
  }

  const source = canvas.tokens.placeables.find(token => token.name === "Caerwyn Thorne")
    ?? canvas.tokens.controlled[0];
  if (!source) {
    return ui.notifications.warn("Place Caerwyn Thorne on the Scene or select one source Token.");
  }

  const checks = [];
  const temporaryRegions = new Set();
  const originalTargets = [...game.user.targets].map(token => token.id);
  let temporaryItem = null;
  const record = (check, passed, details = {}) => checks.push({
    check,
    passed: Boolean(passed),
    details
  });
  const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
  async function waitFor(predicate, label, timeout = 5000) {
    const started = performance.now();
    while (performance.now() - started < timeout) {
      const value = await predicate();
      if (value) return value;
      await wait(50);
    }
    throw new Error(`Timed out waiting for ${label}.`);
  }

  try {
    const descriptor = api.configuration.getCatPropagationConfig();
    record(
      "Public production API and canonical CAT descriptor are available",
      api.configuration.schemaVersion === 1
        && api.configuration.flag === "crosshairs3d"
        && api.configuration.catPropagationKey === "propagation"
        && descriptor.type === "select"
        && descriptor.default === "default"
        && descriptor.options.map(option => option.value).join("|") === "default|none|direct|spread",
      {
        schemaVersion: api.configuration.schemaVersion,
        flag: api.configuration.flag,
        choices: descriptor.options.map(option => option.value)
      }
    );

    const invalid = api.configuration.inspect({
      configuration: {
        schemaVersion: 99,
        shape: {},
        propagation: { mode: "teleport" }
      }
    });
    record(
      "Malformed or unsupported configuration fails before placement",
      invalid.valid === false
        && invalid.errors.some(error => error.includes("schemaVersion"))
        && invalid.errors.some(error => error.includes("shape.type"))
        && invalid.errors.some(error => error.includes("teleport")),
      { errors: invalid.errors }
    );

    const configuration = {
      schemaVersion: 1,
      shape: { type: "sphere", radius: 20 },
      range: { max: 60 },
      capabilities: { elevation: true },
      propagation: { mode: "direct" },
      persistent: {
        enabled: true,
        name: `${TAG} Temporary Sphere`,
        locked: true
      }
    };
    temporaryItem = await Item.create({
      name: `${TAG} Temporary Item`,
      type: "feat",
      system: {},
      flags: {
        [MODULE_ID]: { crosshairs3d: configuration },
        cat: { config: { propagation: "spread" } }
      }
    }, { renderSheet: false });
    if (!temporaryItem) throw new Error("Could not create the temporary World Item.");

    const resolved = await api.configuration.resolve({ item: temporaryItem });
    record(
      "Stored Item configuration survives a document round trip",
      resolved.options.shape.type === "sphere"
        && resolved.options.shape.radius === 20
        && resolved.options.range.max === 60
        && resolved.provenance.sources.includes("item-flag")
        && resolved.provenance.itemUuid === temporaryItem.uuid,
      {
        itemUuid: temporaryItem.uuid,
        sources: resolved.provenance.sources,
        shape: resolved.options.shape
      }
    );
    record(
      "Real CAT Item preference overrides the Item propagation default",
      resolved.provenance.propagation.itemDefault === "direct"
        && resolved.provenance.propagation.override === "spread"
        && resolved.provenance.propagation.mode === "spread"
        && resolved.provenance.propagation.source === "cat-item-configuration",
      resolved.provenance.propagation
    );

    ui.notifications.info(
      `${TAG}: place the 20-foot Sphere, then left-click to confirm it. Right-click would cancel the test.`
    );
    const placed = await api.showConfigured({ source, item: temporaryItem });
    if (placed.persistentRegion?.regionUuid) {
      temporaryRegions.add(placed.persistentRegion.regionUuid);
    }
    const region = placed.persistentRegion?.regionUuid
      ? await fromUuid(placed.persistentRegion.regionUuid)
      : null;
    const metadata = region?.getFlag(MODULE_ID, "crosshair3dPersistentArea");
    record(
      "Configured placement publishes matching provenance and persistence",
      placed.cancelled === false
        && placed.provenance.propagation.mode === "spread"
        && placed.propagation?.mode === "spread"
        && placed.persistentRegion?.created === true
        && metadata?.propagation === "spread",
      {
        cancelled: placed.cancelled,
        propagation: placed.propagation?.mode,
        provenance: placed.provenance.propagation,
        targetIds: placed.targetIds,
        regionUuid: placed.persistentRegion?.regionUuid ?? null,
        backend: metadata?.backend ?? null
      }
    );

    const regionsBeforeCancel = canvas.scene.regions.size;
    const controller = new AbortController();
    const cancellation = api.showConfigured({
      source,
      configuration: {
        schemaVersion: 1,
        shape: { type: "prism", length: 10, width: 10, height: 5 },
        range: { max: 60 },
        propagation: { mode: "none" },
        persistent: { enabled: true, name: `${TAG} Must Not Persist` }
      },
      readCat: false,
      runtime: { signal: controller.signal }
    });
    await waitFor(() => api.getStats().active, "automatic cancellation placement activation");
    controller.abort();
    const cancelled = await cancellation;
    record(
      "Configured cancellation cleans up and creates no persistent Region",
      cancelled.cancelled === true
        && api.getStats().active === false
        && canvas.scene.regions.size === regionsBeforeCancel,
      {
        cancelled: cancelled.cancelled,
        active: api.getStats().active,
        beforeRegions: regionsBeforeCancel,
        afterRegions: canvas.scene.regions.size
      }
    );

    await waitFor(
      () => api.persistent.getAttachedStats().pending === 0,
      "attached-area reload rehydration to become idle"
    );
    const attached = api.persistent.getAttachedStats();
    record(
      "Attached Direct/Spread reload rehydration is initialized and idle",
      attached.initialized === true
        && attached.rehydrations >= 1
        && attached.pending === 0,
      attached
    );

    const configuredStats = api.configured.getStats();
    const recent = api.configured.getRecent();
    record(
      "Configured diagnostics are bounded and include confirmation and cancellation",
      configuredStats.confirmed >= 1
        && configuredStats.cancelled >= 1
        && recent.length <= 20
        && recent.some(entry => entry.type === "confirmed")
        && recent.some(entry => entry.type === "cancelled"),
      {
        configuredStats,
        recentTypes: recent.map(entry => entry.type)
      }
    );

    const passed = checks.every(check => check.passed);
    console.group(TAG);
    console.table(checks.map(row => ({
      check: row.check,
      passed: row.passed,
      details: JSON.stringify(row.details)
    })));
    console.log({ tag: TAG, version: VERSION, passed, checks });
    console.groupEnd();
    ui.notifications[passed ? "info" : "warn"](
      `${TAG}: ${passed ? "PASS" : "CHECK CONSOLE"}.`
    );
    return { tag: TAG, version: VERSION, passed, checks };
  } catch (error) {
    console.error(`${TAG}: FAIL`, error);
    ui.notifications.error(`${TAG}: ${error.message}`);
    return { tag: TAG, version: VERSION, passed: false, error: error.message, checks };
  } finally {
    for (const uuid of temporaryRegions) {
      try {
        const region = await fromUuid(uuid);
        if (!region) continue;
        if (module.api.regions.isOwned(region)) await module.api.regions.delete(uuid);
        else await canvas.scene.deleteEmbeddedDocuments("Region", [region.id]);
      } catch (error) {
        console.warn(`${TAG}: temporary Region cleanup failed`, uuid, error);
      }
    }
    try { await temporaryItem?.delete(); }
    catch (error) { console.warn(`${TAG}: temporary Item cleanup failed`, error); }
    try {
      canvas.tokens.setTargets(
        originalTargets.filter(id => canvas.tokens.get(id)),
        { mode: "replace" }
      );
    } catch (_) { /* cleanup continues */ }
  }
})();
