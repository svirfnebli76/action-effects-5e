await (async () => {
  const TAG = "AE5E 3D Crosshairs Coalescing Acceptance 01";
  const VERSION = "0.4.4.47";
  const MODULE_ID = "action-effects-5e";
  const module = game.modules.get(MODULE_ID);
  const api = module?.api?.crosshairs3d;
  const regionCells = module?.api?.regions?.cells;

  if (!game.user.isGM) return ui.notifications.warn(`${TAG} must be run by the primary GM.`);
  if (module?.version !== VERSION || !api?.show || !api?.getStats || !regionCells?.getConfig) {
    return ui.notifications.warn(`Install and activate AE5E ${VERSION}, then reload Foundry.`);
  }
  if (!canvas.ready || !canvas.grid.isSquare) return ui.notifications.warn("Open a square-grid Scene first.");
  if (api.getStats().active) return ui.notifications.warn("Finish or cancel the current 3D placement first.");

  const source = canvas.tokens.placeables.find(token => token.name === "Caerwyn Thorne")
    ?? canvas.tokens.controlled[0];
  if (!source) return ui.notifications.warn("Place Caerwyn Thorne on the Scene or select one source Token.");

  const checks = [];
  const modeMetrics = {};
  const temporaryRegions = new Set();
  const originalTargets = [...game.user.targets].map(token => token.id);
  const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
  const record = (check, passed, details = {}) => checks.push({ check, passed: Boolean(passed), details });
  const sameIds = (a, b) => [...a].sort().join("|") === [...b].sort().join("|");
  const cellKey = cell => `${cell.x},${cell.y},${cell.z}`;
  const delta = (after, before, key) => Number(after?.[key] ?? 0) - Number(before?.[key] ?? 0);

  const restoreTargets = () => {
    const ids = originalTargets.filter(id => canvas.tokens.get(id));
    canvas.tokens.setTargets(ids, { mode: "replace" });
  };

  const percentile = (values, p) => {
    if (!values.length) return null;
    const sorted = [...values].sort((a, b) => a - b);
    const index = Math.min(sorted.length - 1, Math.max(0, Math.ceil((p / 100) * sorted.length) - 1));
    return sorted[index];
  };

  const frameSummary = values => {
    const clean = values.filter(value => Number.isFinite(value) && value >= 0 && value < 1000);
    return {
      samples: clean.length,
      medianMs: percentile(clean, 50),
      p95Ms: percentile(clean, 95),
      maxMs: clean.length ? Math.max(...clean) : null,
      over33ms: clean.filter(value => value > 33.4).length,
      over50ms: clean.filter(value => value > 50).length
    };
  };

  async function waitFor(predicate, label, timeout = 5000) {
    const started = performance.now();
    while (performance.now() - started < timeout) {
      const value = await predicate();
      if (value) return value;
      await wait(25);
    }
    throw new Error(`Timed out waiting for ${label}.`);
  }

  async function deleteRegionUuid(uuid) {
    if (!uuid) return;
    const region = await fromUuid(uuid).catch(() => null);
    if (!region) return;
    if (typeof region.delete === "function") await region.delete();
    else await canvas.scene.deleteEmbeddedDocuments("Region", [region.id]);
  }

  async function runInteractive(mode, { persistent = false } = {}) {
    restoreTargets();

    const before = api.getStats();
    const authoritative = [];
    const frameIntervals = [];
    let frameWatching = true;
    let lastFrame = null;

    const frameDone = new Promise(resolve => {
      const tick = timestamp => {
        if (!frameWatching) return resolve();
        if (lastFrame !== null) frameIntervals.push(timestamp - lastFrame);
        lastFrame = timestamp;
        requestAnimationFrame(tick);
      };
      requestAnimationFrame(tick);
    });

    ui.notifications.info(
      `${TAG}: ${mode.toUpperCase()} — move the 20-foot Sphere continuously for several seconds, then left-click to confirm.`
    );

    let result;
    try {
      result = await api.show({
        source,
        shape: { type: "sphere", radius: 20 },
        range: { max: 60 },
        capabilities: { elevation: true },
        propagation: { override: mode },
        persistent: persistent
          ? { enabled: true, name: `${TAG} Temporary ${mode} Sphere`, locked: true }
          : { enabled: false },
        onRevision: revision => authoritative.push({
          serial: revision.serial,
          targetIds: revision.targets.map(token => token.id)
        })
      });
    } finally {
      frameWatching = false;
      await frameDone;
    }

    const after = api.getStats();
    const lastAuthoritative = authoritative.at(-1) ?? null;
    const liveTargetIds = [...game.user.targets].map(token => token.id);
    const serialsMonotonic = authoritative.every(
      (entry, index) => index === 0 || entry.serial > authoritative[index - 1].serial
    );

    const metrics = {
      presentations: delta(after, before, "presentations"),
      authoritativeRevisions: delta(after, before, "revisions"),
      propagationStarts: delta(after, before, "propagationStarts"),
      coalescedRequests: delta(after, before, "coalescedRequests"),
      staleDiscards: delta(after, before, "staleDiscards"),
      frames: frameSummary(frameIntervals)
    };
    modeMetrics[mode] = metrics;

    record(
      `${mode}: final authoritative revision matches the confirmed visual request`,
      result.cancelled === false
        && lastAuthoritative?.serial === result.revision.serial
        && serialsMonotonic,
      {
        resultSerial: result.revision?.serial ?? null,
        lastAuthoritativeSerial: lastAuthoritative?.serial ?? null,
        authoritativeSerials: authoritative.map(entry => entry.serial),
        ...metrics
      }
    );

    if (mode !== "none") {
      record(
        `${mode}: rapid visual intent is coalesced before physical propagation starts`,
        metrics.coalescedRequests > 0
          && metrics.presentations > 0
          && metrics.propagationStarts < metrics.presentations,
        metrics
      );
    }

    record(
      `${mode}: only the final authoritative targets remain published`,
      result.cancelled === false
        && sameIds(liveTargetIds, result.targetIds ?? [])
        && sameIds(lastAuthoritative?.targetIds ?? [], result.targetIds ?? []),
      {
        liveTargetIds,
        resultTargetIds: result.targetIds ?? [],
        lastAuthoritativeTargetIds: lastAuthoritative?.targetIds ?? []
      }
    );

    if (persistent && result.persistentRegion?.regionUuid) {
      temporaryRegions.add(result.persistentRegion.regionUuid);
      const region = await fromUuid(result.persistentRegion.regionUuid);
      const config = regionCells.getConfig(region);
      const metadata = region?.getFlag(MODULE_ID, "crosshair3dPersistentArea");
      const expectedKeys = (result.propagation?.cells ?? []).map(cellKey).sort();
      const storedKeys = Object.entries(config?.cells ?? {})
        .filter(([, state]) => String(state).toUpperCase() === "ACTIVE")
        .map(([key]) => key)
        .sort();

      record(
        `${mode}: persistent Region stores the exact final propagated cell set`,
        result.persistentRegion?.created === true
          && result.propagation?.mode === mode
          && metadata?.propagation === mode
          && expectedKeys.join("|") === storedKeys.join("|"),
        {
          regionUuid: result.persistentRegion.regionUuid,
          expectedCells: expectedKeys.length,
          storedCells: storedKeys.length,
          propagation: result.propagation?.mode ?? null,
          metadataPropagation: metadata?.propagation ?? null
        }
      );
    }

    restoreTargets();
    return result;
  }

  try {
    const none = await runInteractive("none");
    const direct = await runInteractive("direct");
    const spread = await runInteractive("spread", { persistent: true });

    record(
      "None, Direct, and Spread all completed with the same 20-foot Sphere placement contract",
      [none, direct, spread].every(result =>
        result.cancelled === false
        && result.shape?.type === "sphere"
        && result.shape?.radius === 20
        && ["none", "direct", "spread"].includes(result.propagation?.mode)
      ),
      {
        modes: [none.propagation?.mode, direct.propagation?.mode, spread.propagation?.mode],
        cells: [
          none.propagation?.cells?.length,
          direct.propagation?.cells?.length,
          spread.propagation?.cells?.length
        ]
      }
    );

    restoreTargets();
    const regionsBeforeCancel = canvas.scene.regions.size;
    const controller = new AbortController();
    const cancellation = api.show({
      source,
      shape: { type: "sphere", radius: 20 },
      range: { max: 60 },
      propagation: { override: "spread" },
      persistent: { enabled: true, name: `${TAG} Must Not Persist` },
      signal: controller.signal
    });

    await waitFor(() => api.getStats().active, "cancellation placement activation");
    controller.abort();
    const cancelled = await cancellation;

    record(
      "Cancellation invalidates pending work, restores targets, and creates no Region",
      cancelled.cancelled === true
        && api.getStats().active === false
        && canvas.scene.regions.size === regionsBeforeCancel
        && sameIds([...game.user.targets].map(token => token.id), originalTargets),
      {
        cancelled: cancelled.cancelled,
        active: api.getStats().active,
        beforeRegions: regionsBeforeCancel,
        afterRegions: canvas.scene.regions.size,
        targetIds: [...game.user.targets].map(token => token.id)
      }
    );

    const passed = checks.every(check => check.passed);

    console.group(`${TAG} — ${passed ? "PASS" : "FAIL"}`);
    console.table(checks.map(entry => ({
      check: entry.check,
      passed: entry.passed,
      details: JSON.stringify(entry.details)
    })));
    console.table(Object.entries(modeMetrics).map(([mode, metrics]) => ({
      mode,
      presentations: metrics.presentations,
      propagationStarts: metrics.propagationStarts,
      coalescedRequests: metrics.coalescedRequests,
      staleDiscards: metrics.staleDiscards,
      authoritativeRevisions: metrics.authoritativeRevisions,
      frameMedianMs: metrics.frames.medianMs,
      frameP95Ms: metrics.frames.p95Ms,
      frameMaxMs: metrics.frames.maxMs,
      framesOver33Ms: metrics.frames.over33ms,
      framesOver50Ms: metrics.frames.over50ms
    })));
    console.log({
      tag: TAG,
      version: VERSION,
      passed,
      checks,
      modeMetrics,
      finalStats: api.getStats(),
      visualAcceptanceRequired: true,
      visualQuestion: "Did Direct and Spread now feel essentially as smooth as None during continuous movement?"
    });
    console.groupEnd();

    ui.notifications[passed ? "info" : "error"](
      `${TAG}: ${passed ? "objective checks PASS" : "one or more objective checks FAILED"}. `
      + "Please report whether Direct and Spread now feel essentially as smooth as None."
    );

    return { tag: TAG, version: VERSION, passed, checks, modeMetrics };
  } catch (error) {
    console.error(`${TAG} failed`, error);
    ui.notifications.error(`${TAG} failed: ${error.message}`);
    throw error;
  } finally {
    restoreTargets();
    for (const uuid of temporaryRegions) {
      try {
        await deleteRegionUuid(uuid);
      } catch (error) {
        console.warn(`${TAG}: temporary Region cleanup failed`, uuid, error);
      }
    }
  }
})();
