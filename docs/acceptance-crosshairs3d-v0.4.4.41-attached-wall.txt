await (async () => {
  const TAG = "AE5E v0.4.4.41 Attached Wall Acceptance 01";
  const VERSION = "0.4.4.41";
  const MODULE_ID = "action-effects-5e";
  const ae5eModule = game.modules.get(MODULE_ID);
  const api = ae5eModule?.api?.crosshairs3d;

  if (!game.user.isGM) return ui.notifications.warn(`${TAG} must be run by the primary GM.`);
  if (ae5eModule?.version !== VERSION || !api?.persistent?.getAttachedStats) {
    return ui.notifications.warn(`Install and activate AE5E ${VERSION}, then reload Foundry.`);
  }
  if (!canvas.ready || !canvas.grid.isSquare) {
    return ui.notifications.warn("Open a square-grid Scene first.");
  }

  const source = canvas.tokens.placeables.find(token => token.name === "Caerwyn Thorne")
    ?? canvas.tokens.controlled[0];
  if (!source) return ui.notifications.warn("Place Caerwyn Thorne on the Scene or select one source Token.");

  const scene = canvas.scene;
  const rect = canvas.dimensions.sceneRect;
  const size = scene.grid.size;
  const distance = scene.grid.distance;
  const metrics = Object.freeze({ size, distance, originX: rect.x, originY: rect.y,
    grid: Object.freeze({ distance, origin: Object.freeze({ x: 0, y: 0, z: 0 }) }) });
  const snapshot = { x: source.document.x, y: source.document.y,
    elevation: source.document.elevation, rotation: source.document.rotation };
  let regionUuid = null;
  let wallId = null;
  const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
  const completed = stats => stats.resolved + stats.deactivated + stats.errors;
  const storedCells = region => Object.keys(region?.getFlag(MODULE_ID, "regionCells")?.cells ?? {}).length;

  async function waitForHook(previous, label, timeout = 6000) {
    const started = performance.now();
    while (performance.now() - started < timeout) {
      const stats = api.persistent.getAttachedStats();
      if (completed(stats) > previous && stats.pending === 0) return stats;
      await wait(50);
    }
    throw new Error(`Timed out waiting for ${label}.`);
  }

  async function waitForIdle(timeout = 6000) {
    const started = performance.now();
    while (performance.now() - started < timeout) {
      if (api.persistent.getAttachedStats().pending === 0) return;
      await wait(50);
    }
    throw new Error("Timed out waiting for attached-area cleanup.");
  }

  const toDistance = (x, y, z) => ({
    x: ((x - rect.x) / size) * distance,
    y: ((y - rect.y) / size) * distance,
    z
  });
  const toPixels = point => ({
    x: rect.x + (point.x / distance * size),
    y: rect.y + (point.y / distance * size)
  });

  try {
    await source.document.update({ x: 2500, y: 2500, elevation: 0, rotation: 0 }, { animate: false });
    const origin = toDistance(
      source.document.x + source.document.width * size / 2,
      source.document.y + source.document.height * size / 2,
      source.document.elevation
    );
    const shape = api.geometry.normalizeShape({
      type: "sphere",
      origin: { x: origin.x + 4 * distance, y: origin.y, z: origin.z },
      radius: 2 * distance
    });
    const resolveRaw = () => api.propagation.apply({
      shape,
      grid: metrics.grid,
      mode: "direct",
      origin,
      environment: api.propagation.createFoundryEnvironment({ scene, source, metrics })
    });

    const clearPropagation = await resolveRaw();
    if (!clearPropagation.cells.length) throw new Error("The fixture is obstructed before the test Wall is created.");
    const created = await api.persistent.create({
      propagation: clearPropagation,
      scene,
      source,
      metrics,
      options: { name: `${TAG} Temporary Area`, attached: true, locked: true }
    });
    regionUuid = created.regionUuid;
    const clearStored = storedCells(await fromUuid(regionUuid));

    const wallX = origin.x + 2 * distance;
    const a = toPixels({ x: wallX, y: origin.y - 3 * distance });
    const b = toPixels({ x: wallX, y: origin.y + 3 * distance });
    const normal = CONST.EDGE_SENSE_TYPES?.NORMAL ?? CONST.WALL_SENSE_TYPES?.NORMAL ?? 1;
    const statsBeforeCreate = api.persistent.getAttachedStats();
    const errorsBefore = statsBeforeCreate.errors;
    const [wall] = await scene.createEmbeddedDocuments("Wall", [{
      c: [a.x, a.y, b.x, b.y], move: normal, sight: 0, light: 0, sound: 0, door: 0, ds: 0
    }]);
    wallId = wall.id;
    const blockedStats = await waitForHook(completed(statsBeforeCreate), "Wall-create re-propagation");
    const blockedRaw = (await resolveRaw()).cells.length;
    const blockedStored = storedCells(await fromUuid(regionUuid));

    const beforeDelete = completed(blockedStats);
    await scene.deleteEmbeddedDocuments("Wall", [wallId]);
    wallId = null;
    const restoredStats = await waitForHook(beforeDelete, "Wall-delete re-propagation");
    const restoredRaw = (await resolveRaw()).cells.length;
    const restoredStored = storedCells(await fromUuid(regionUuid));

    const counts = {
      clearRaw: clearPropagation.cells.length,
      clearStored,
      blockedRaw,
      blockedStored,
      restoredRaw,
      restoredStored
    };
    const passed = clearStored === counts.clearRaw
      && blockedRaw < counts.clearRaw
      && blockedStored === blockedRaw
      && restoredRaw === counts.clearRaw
      && restoredStored === counts.clearRaw
      && restoredStats.errors === errorsBefore;
    const result = { tag: TAG, version: VERSION, passed, counts,
      reasons: { blocked: blockedStats.last?.reason ?? null,
        restored: restoredStats.last?.reason ?? null }, stats: restoredStats };

    console.group(TAG);
    console.table([counts]);
    console.log(result);
    console.groupEnd();
    ui.notifications[passed ? "info" : "warn"](
      `${TAG}: ${passed ? "PASS" : "CHECK CONSOLE"}. Stored mask ${clearStored} → ${blockedStored} → ${restoredStored}.`
    );
    return result;
  } catch (error) {
    console.error(`${TAG}: FAIL`, error);
    ui.notifications.error(`${TAG}: ${error.message}`);
    return { tag: TAG, version: VERSION, passed: false, error: error.message };
  } finally {
    try {
      if (wallId && scene.walls.get(wallId)) {
        const before = completed(api.persistent.getAttachedStats());
        await scene.deleteEmbeddedDocuments("Wall", [wallId]);
        if (regionUuid) await waitForHook(before, "cleanup Wall deletion");
      } else await waitForIdle();
    } catch (error) { console.warn(`${TAG}: temporary Wall cleanup failed`, error); }
    try {
      const region = regionUuid ? await fromUuid(regionUuid) : null;
      if (region) await ae5eModule.api.regions.delete(regionUuid);
    } catch (error) { console.warn(`${TAG}: temporary Region cleanup failed`, error); }
    try { await source.document.update(snapshot, { animate: false }); }
    catch (error) { console.warn(`${TAG}: source restoration failed`, error); }
  }
})();
