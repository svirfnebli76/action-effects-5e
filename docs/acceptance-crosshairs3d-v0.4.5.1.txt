await (async () => {
  const TAG = "AE5E 3D Crosshairs Target Preview Notice Acceptance 01";
  const VERSION = "0.4.5.1";
  const MODULE_ID = "action-effects-5e";
  const NOTICE = "Target Preview takes 1-2sec";
  const module = game.modules.get(MODULE_ID);
  const api = module?.api?.crosshairs3d;

  if (!game.user.isGM) return ui.notifications.warn(`${TAG} must be run by the primary GM.`);
  if (module?.version !== VERSION || !api?.show || !api?.getStats) {
    return ui.notifications.warn(`Install and activate AE5E ${VERSION}, then reload Foundry.`);
  }
  if (!canvas.ready || !canvas.grid.isSquare) return ui.notifications.warn("Open a square-grid Scene first.");
  if (api.getStats().active) return ui.notifications.warn("Finish or cancel the current 3D placement first.");

  const source = canvas.tokens.placeables.find(token => token.name === "Caerwyn Thorne")
    ?? canvas.tokens.controlled[0];
  if (!source) return ui.notifications.warn("Place Caerwyn Thorne on the Scene or select one source Token.");

  const checks = [];
  const originalTargets = [...game.user.targets].map(token => token.id);
  const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
  const record = (check, passed, details = {}) => checks.push({ check, passed: Boolean(passed), details });

  const restoreTargets = () => {
    const ids = originalTargets.filter(id => canvas.tokens.get(id));
    canvas.tokens.setTargets(ids, { mode: "replace" });
  };

  async function waitFor(predicate, label, timeout = 5000) {
    const started = performance.now();
    while (performance.now() - started < timeout) {
      const value = predicate();
      if (value) return value;
      await wait(25);
    }
    throw new Error(`Timed out waiting for ${label}.`);
  }

  function collectText(node, output = []) {
    if (!node) return output;
    if (typeof node.text === "string") output.push(node);
    for (const child of node.children ?? []) collectText(child, output);
    return output;
  }

  async function runMode(mode) {
    restoreTargets();
    const expected = mode !== "none";
    ui.notifications.info(
      `${TAG}: ${mode.toUpperCase()} — ${expected ? `confirm that "${NOTICE}" is visible below the control hint` : `confirm that "${NOTICE}" is absent`}, then left-click.`
    );

    const placement = api.show({
      source,
      shape: { type: "sphere", radius: 20 },
      range: { max: 60 },
      capabilities: { elevation: true },
      propagation: { override: mode },
      persistent: { enabled: false }
    });

    await waitFor(() => api.getStats().active, `${mode} placement activation`);
    await wait(50);

    const root = canvas.interface.children.find(child => child.name === "ae5e-crosshairs3d-placement");
    const labels = collectText(root);
    const noticeLabels = labels.filter(label => label.text === NOTICE);
    const controlLabels = labels.filter(label => label.text === "CTRL");

    record(
      `${mode}: target-preview notice visibility matches propagation mode`,
      expected ? noticeLabels.length === 1 : noticeLabels.length === 0,
      {
        expected,
        noticeCount: noticeLabels.length,
        controlHintCount: controlLabels.length
      }
    );

    if (expected && noticeLabels[0] && controlLabels[0]) {
      record(
        `${mode}: target-preview notice uses the same font size as the control hint`,
        Number(noticeLabels[0].style?.fontSize) === Number(controlLabels[0].style?.fontSize),
        {
          noticeFontSize: noticeLabels[0].style?.fontSize ?? null,
          controlFontSize: controlLabels[0].style?.fontSize ?? null
        }
      );
    }

    const result = await placement;
    record(
      `${mode}: placement confirms cleanly`,
      result.cancelled === false,
      { cancelled: result.cancelled }
    );
    restoreTargets();
  }

  try {
    await runMode("none");
    await runMode("direct");
    await runMode("spread");

    const passed = checks.every(check => check.passed);
    console.group(`${TAG} — ${passed ? "PASS" : "FAIL"}`);
    console.table(checks.map(entry => ({
      check: entry.check,
      passed: entry.passed,
      details: JSON.stringify(entry.details)
    })));
    console.log({
      tag: TAG,
      version: VERSION,
      passed,
      checks,
      visualAcceptanceRequired: true,
      visualQuestion: `Is "${NOTICE}" centered directly below the existing control hint, same size, color #e0dcdd, and absent in None?`
    });
    console.groupEnd();

    ui.notifications[passed ? "info" : "error"](
      `${TAG}: ${passed ? "objective checks PASS" : "one or more objective checks FAILED"}. Please confirm the visual placement and color.`
    );

    return { tag: TAG, version: VERSION, passed, checks };
  } catch (error) {
    console.error(`${TAG} failed`, error);
    ui.notifications.error(`${TAG} failed: ${error.message}`);
    throw error;
  } finally {
    restoreTargets();
  }
})();
