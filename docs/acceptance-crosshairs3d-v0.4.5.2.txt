await (async () => {
  const TAG = "AE5E 3D Crosshairs Line/Cone Propagation Acceptance 01";
  const VERSION = "0.4.5.2";
  const MODULE_ID = "action-effects-5e";

  const module = game.modules.get(MODULE_ID);
  const api = module?.api?.crosshairs3d;
  const source = canvas.tokens?.controlled?.[0]
    ?? canvas.tokens.placeables.find(token => token.name === "Caerwyn Thorne");

  if (!game.user.isGM) {
    return ui.notifications.warn(`${TAG} must be run by the primary GM.`);
  }
  if (
    module?.version !== VERSION
    || !api?.show
    || !api?.getStats
    || !api?.cells?.rasterize
    || !api?.propagation?.apply
    || !api?.propagation?.createFoundryEnvironment
  ) {
    return ui.notifications.warn(`Install AE5E v${VERSION} and reload Foundry first.`);
  }
  if (!canvas.ready || !canvas.grid.isSquare) {
    return ui.notifications.warn("Open a square-grid Scene first.");
  }
  if (!source) {
    return ui.notifications.warn("Select a source Token first.");
  }
  if (api.getStats().active) {
    return ui.notifications.warn("Finish or cancel the current 3D placement first.");
  }

  const choice = await foundry.applications.api.DialogV2.wait({
    window: { title: `AE5E ${VERSION} — Line/Cone Propagation Acceptance` },
    position: { width: 520 },
    content: `
      <p>This test first runs automatic clear-space geometry checks, then opens four live placements:</p>
      <ol>
        <li>Source-driven Line — Spread</li>
        <li>Freely Placed Line — Spread</li>
        <li>Cone — Direct</li>
        <li>Cone — Spread</li>
      </ol>
      <p>For the Line tests, aim at a diagonal/non-cardinal angle. For the Cone tests, aim upward so the 3D Cone visibly contains an elevated Token. Allow 1–2 seconds for the target preview before confirming.</p>
      <p>No Items, Walls, or Regions are created. Your original target selection is restored at the end.</p>
    `,
    buttons: [
      { action: "start", label: "Start Acceptance", default: true },
      { action: "cancel", label: "Cancel" }
    ],
    rejectClose: false
  });

  if (choice !== "start") return;

  const checks = [];
  const liveResults = [];
  const originalTargets = [...game.user.targets].map(token => token.id);
  const grid = {
    distance: Number(canvas.scene.grid.distance),
    origin: { x: 0, y: 0, z: 0 }
  };

  const key = cell => `${cell.x},${cell.y},${cell.z}`;
  const keys = cells => [...cells].map(key).sort();
  const sameMask = (a, b) => keys(a).join("|") === keys(b).join("|");
  const record = (check, passed, details = {}) => checks.push({
    check,
    passed: Boolean(passed),
    details
  });

  const restoreTargets = () => {
    const ids = originalTargets.filter(id => canvas.tokens.get(id));
    canvas.tokens.setTargets(ids, { mode: "replace" });
  };

  const clearSpread = {
    seedOpen: () => true,
    sharedFace: () => ({ largestContiguousFraction: 1 })
  };

  const clearDirect = api.propagation.createFoundryEnvironment({
    scene: canvas.scene,
    source,
    collision: () => false
  });

  async function verifySpread(label, shape) {
    const expected = api.cells.rasterize(shape, { grid }).cells;
    const result = await api.propagation.apply({
      shape,
      grid,
      mode: "spread",
      environment: clearSpread
    });

    record(
      `${label}: clear-space Spread reaches every authoritative affected cell`,
      sameMask(expected, result.cells),
      {
        candidateCells: expected.length,
        spreadCells: result.cells.length,
        seedQueries: result.stats?.seedQueries ?? null,
        faceQueries: result.stats?.faceQueries ?? null
      }
    );
  }

  async function verifyConeDirect(pitch) {
    const shape = {
      type: "cone",
      origin: { x: 0, y: 0, z: 0 },
      length: 15,
      yaw: 0,
      pitch
    };
    const expected = api.cells.rasterize(shape, { grid }).cells;
    const result = await api.propagation.apply({
      shape,
      grid,
      mode: "direct",
      environment: clearDirect
    });

    record(
      `Cone pitch ${pitch}°: clear-space Direct preserves every normally qualifying cell`,
      sameMask(expected, result.cells),
      {
        candidateCells: expected.length,
        directCells: result.cells.length,
        directQueries: result.stats?.directQueries ?? null
      }
    );
  }

  async function runLiveCase({ label, mode, shape, range, instruction }) {
    restoreTargets();

    ui.notifications.info(
      `${label} / ${mode.toUpperCase()}: ${instruction}`
    );

    const result = await api.show({
      source,
      shape,
      range,
      propagation: { override: mode },
      persistent: { enabled: false }
    });

    const stats = api.getStats();
    const resolvedMode = result.propagation?.mode ?? null;
    const targetIds = result.targetIds ?? result.targets?.map(token => token.id) ?? [];

    const passed =
      result.cancelled === false
      && !stats.active
      && resolvedMode === mode;

    record(
      `${label}: live ${mode} placement confirms with the requested propagation mode`,
      passed,
      {
        cancelled: result.cancelled,
        resolvedMode,
        targetIds,
        propagationCells: result.propagation?.cells?.length ?? null
      }
    );

    liveResults.push({
      label,
      mode,
      targetIds,
      propagationCells: result.propagation?.cells?.length ?? null,
      cancelled: result.cancelled
    });

    restoreTargets();
  }

  try {
    for (const yaw of [15, 30, 45, 60, 75]) {
      await verifySpread(
        `Source Line yaw ${yaw}°`,
        {
          type: "line",
          origin: { x: 0, y: 0, z: 0 },
          length: 60,
          width: 5,
          yaw,
          pitch: 0
        }
      );
    }

    for (const pitch of [15, 30, 45, 60]) {
      await verifySpread(
        `Source Line pitch ${pitch}°`,
        {
          type: "line",
          origin: { x: 0, y: 0, z: 0 },
          length: 60,
          width: 5,
          yaw: 0,
          pitch
        }
      );
    }

    for (const yaw of [15, 30, 45, 60, 75]) {
      await verifySpread(
        `Free Line yaw ${yaw}°`,
        {
          type: "free-line",
          origin: { x: 0, y: 0, z: 0 },
          length: 60,
          width: 5,
          height: 20,
          yaw
        }
      );
    }

    for (const pitch of [15, 30, 45, 60, 75, 90]) {
      await verifySpread(
        `Cone pitch ${pitch}°`,
        {
          type: "cone",
          origin: { x: 0, y: 0, z: 0 },
          length: 15,
          yaw: 0,
          pitch
        }
      );
      await verifyConeDirect(pitch);
    }

    await runLiveCase({
      label: "Source-driven Line",
      mode: "spread",
      shape: { type: "line", length: 60, width: 5 },
      range: { max: 60 },
      instruction:
        "Aim diagonally through one or more Tokens, wait 1–2 sec for Target Preview, then left-click. Confirm the preview does not disappear simply because the Line is diagonal."
    });

    await runLiveCase({
      label: "Freely Placed Line",
      mode: "spread",
      shape: { type: "free-line", length: 60, width: 5, height: 20 },
      range: { max: 120, policy: "endpoints" },
      instruction:
        "Place/rotate it diagonally through one or more Tokens, wait 1–2 sec for Target Preview, then left-click. Confirm the preview remains stable."
    });

    await runLiveCase({
      label: "Cone",
      mode: "direct",
      shape: { type: "cone", length: 15 },
      range: { max: 60 },
      instruction:
        "Aim the Cone upward so its 3D volume contains an elevated Token, wait 1–2 sec for Target Preview, then left-click. Confirm the elevated Token stays targeted."
    });

    await runLiveCase({
      label: "Cone",
      mode: "spread",
      shape: { type: "cone", length: 15 },
      range: { max: 60 },
      instruction:
        "Aim the Cone upward so its 3D volume contains an elevated Token, wait 1–2 sec for Target Preview, then left-click. Confirm the elevated Token stays targeted."
    });

    const passed = checks.every(check => check.passed);

    console.group(`${TAG} — ${passed ? "PASS" : "FAIL"}`);
    console.table(
      checks.map(entry => ({
        check: entry.check,
        passed: entry.passed,
        details: JSON.stringify(entry.details)
      }))
    );
    console.table(liveResults);
    console.log({
      tag: TAG,
      version: VERSION,
      passed,
      checks,
      liveResults,
      visualAcceptanceRequired: true,
      visualQuestions: [
        "Did Source-driven Line Spread keep stable targeting at diagonal angles?",
        "Did Freely Placed Line Spread keep stable targeting at diagonal angles?",
        "Did Cone Direct target elevated Tokens inside the visible 3D Cone?",
        "Did Cone Spread target elevated Tokens inside the visible 3D Cone?"
      ]
    });
    console.groupEnd();

    ui.notifications[passed ? "info" : "error"](
      `${TAG}: objective checks ${passed ? "PASS" : "FAIL"}. `
      + "Please report the four visual targeting results."
    );

    return {
      tag: TAG,
      version: VERSION,
      passed,
      checks,
      liveResults
    };
  } catch (error) {
    console.error(`${TAG} failed`, error);
    ui.notifications.error(`${TAG} failed: ${error.message}`);
    throw error;
  } finally {
    restoreTargets();
  }
})();
