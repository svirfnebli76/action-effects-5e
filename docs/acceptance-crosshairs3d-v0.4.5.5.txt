await (async () => {
  const TAG = "AE5E Source-Bound Prism Acceptance 01";
  const VERSION = "0.4.5.5";
  const MODULE_ID = "action-effects-5e";

  const module = game.modules.get(MODULE_ID);
  const api = module?.api?.crosshairs3d;
  const source = canvas.tokens?.controlled?.[0]
    ?? canvas.tokens.placeables.find(token => token.name === "Caerwyn Thorne");

  if (!game.user.isGM) {
    return ui.notifications.warn(`${TAG} must be run by the primary GM.`);
  }
  if (module?.version !== VERSION || !api?.show || !api?.getStats) {
    return ui.notifications.warn(`Install AE5E v${VERSION} and reload Foundry first.`);
  }
  if (!canvas.ready || !canvas.grid.isSquare) {
    return ui.notifications.warn("Open a square-grid Scene first.");
  }
  if (!source) {
    return ui.notifications.warn("Select a source Token first.");
  }
  if (api.getStats().active) {
    return ui.notifications.warn("Finish or cancel the current 3D placement first.");
  }

  const choice = await foundry.applications.api.DialogV2.wait({
    window: { title: `AE5E ${VERSION} — Source-Bound Prism Acceptance` },
    position: { width: 590 },
    content: `
      <p>This acceptance opens the new 15 × 15 × 15 ft source-bound Prism three times:</p>
      <ol>
        <li><b>None</b></li>
        <li><b>Direct</b></li>
        <li><b>Spread</b></li>
      </ol>
      <p>For each placement, move the cursor around all four sides of the selected Token and verify the cube remains attached continuously to the Token edge while extending fully outward.</p>
      <p>Only MOVE should be available. SHIFT+wheel, CTRL+wheel, and ALT+wheel should not manipulate the cube or appear as control hints.</p>
      <p>Direct and Spread should show <b>Target Preview takes 1-2sec</b>; None should not. Confirm each placement with left-click.</p>
    `,
    buttons: [
      { action: "start", label: "Start Acceptance", default: true },
      { action: "cancel", label: "Cancel" }
    ],
    rejectClose: false
  });
  if (choice !== "start") return;

  const originalTargets = [...game.user.targets].map(token => token.id);
  const results = [];
  const restoreTargets = () => {
    const ids = originalTargets.filter(id => canvas.tokens.get(id));
    canvas.tokens.setTargets(ids, { mode: "replace" });
  };

  const runCase = async mode => {
    restoreTargets();
    ui.notifications.info(
      `${TAG}: ${mode.toUpperCase()} — move around the source, verify MOVE-only controls, then confirm.`
    );

    const placed = await api.show({
      source,
      shape: { type: "prism", length: 15, width: 15, height: 15 },
      placement: { mode: "source" },
      capabilities: { elevation: false, rotation: false, resize: false },
      propagation: { itemDefault: mode, override: "default" },
      persistent: { enabled: false }
    });

    if (placed.cancelled) {
      results.push({ mode, passed: false, reason: "cancelled" });
      return false;
    }

    const anchor = placed.sourceAnchor;
    const center = placed.placementPoint;
    const halfLength = 7.5;
    const anchorDistance = anchor && center
      ? Math.hypot(center.x - anchor.x, center.y - anchor.y)
      : NaN;
    const modeOk = placed.propagation?.mode === mode;
    const anchorOk = Number.isFinite(anchorDistance)
      && Math.abs(anchorDistance - halfLength) < 1e-6;
    const clean = api.getStats().active === false;
    const passed = modeOk && anchorOk && clean;

    results.push({
      mode,
      passed,
      propagationMode: placed.propagation?.mode ?? null,
      sourceAnchor: anchor ?? null,
      placementPoint: center ?? null,
      centerFromAnchor: anchorDistance,
      targetIds: placed.targetIds ?? [],
      propagationCells: placed.propagation?.cells?.length ?? null,
      sessionClean: clean
    });
    restoreTargets();
    return passed;
  };

  try {
    for (const mode of ["none", "direct", "spread"]) {
      const completed = await runCase(mode);
      if (!completed && results.at(-1)?.reason === "cancelled") break;
    }
  } finally {
    restoreTargets();
  }

  const automaticPass = results.length === 3 && results.every(result => result.passed);
  console.group(`${TAG} — Results`);
  console.table(results.map(result => ({
    mode: result.mode,
    passed: result.passed,
    propagationMode: result.propagationMode ?? result.reason ?? "",
    centerFromAnchor: Number.isFinite(result.centerFromAnchor)
      ? result.centerFromAnchor.toFixed(3)
      : "",
    propagationCells: result.propagationCells ?? "",
    targets: result.targetIds?.length ?? "",
    sessionClean: result.sessionClean ?? ""
  })));
  console.log("Full results:", results);
  console.groupEnd();

  if (automaticPass) {
    ui.notifications.info(
      `${TAG}: automatic checks PASS. If the three visual/control checks also looked correct, live acceptance is complete.`
    );
  } else {
    ui.notifications.warn(`${TAG}: one or more automatic checks did not pass. See console results.`);
  }
})();
