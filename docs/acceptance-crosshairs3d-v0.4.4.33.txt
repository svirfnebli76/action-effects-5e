await (async () => {
  const module = game.modules.get("action-effects-5e");
  const api = module?.api?.crosshairs3d;
  const source = canvas.tokens?.controlled[0];
  if (module?.version !== "0.4.4.33" || !api?.show) {
    return ui.notifications.warn("Install AE5E v0.4.4.33 and reload first.");
  }
  if (!source || !canvas.ready) return ui.notifications.warn("Select a source token first.");
  const cases = {
    line: { label: "Source-driven line", shape: { type: "line", length: 60, width: 5 } },
    free: { label: "Freely placed line", shape: { type: "free-line", length: 60, width: 5, height: 20 }, range: { max: 120, policy: "endpoints" } },
    sphere: { label: "Sphere", shape: { type: "sphere", radius: 10 }, range: { max: 60 } },
    prism: { label: "Prism / cube", shape: { type: "prism", length: 20, width: 20, height: 20 }, range: { max: 60 } },
    ground: { label: "Shallow prism (20 × 20 × 5)", shape: { type: "prism", length: 20, width: 20, height: 5 }, range: { max: 60 } },
    cone: { label: "Cone", shape: { type: "cone", length: 15 } },
    cylinder: { label: "Cylinder", shape: { type: "cylinder", radius: 10, height: 20 }, range: { max: 60 } }
  };
  const choice = await foundry.applications.api.DialogV2.wait({
    window: { title: "AE5E 0.4.4.33 — PIXI Crosshair Acceptance" },
    position: { width: 440 },
    content: "<p>Select a shape. Left-click confirms; right-click or Escape cancels. No Regions or Items are created.</p>",
    buttons: Object.entries(cases).map(([action, value]) => ({ action, label: value.label })),
    rejectClose: false
  });
  const selected = cases[choice];
  if (!selected) return;
  try {
    const result = await api.show({ source, shape: selected.shape, range: selected.range });
    const passed = !api.getStats().active && game.modules.get("action-effects-5e").api.crosshairs === undefined;
    console.log(`AE5E PIXI CROSSHAIRS — ${passed ? "PASS" : "FAIL"} (lifecycle only)`, { shape: selected.label, result, stats: api.getStats() });
    if (!passed) throw new Error("Placement session did not clean up or legacy API is still present.");
    ui.notifications.info(`${selected.label}: ${result.cancelled ? "cancelled" : "confirmed"}; lifecycle PASS. Please verify appearance and targets visually.`);
    return result;
  } catch (error) {
    console.error("AE5E PIXI CROSSHAIRS — FAIL", error);
    ui.notifications.error(`AE5E crosshair FAIL: ${error.message}`);
    throw error;
  }
})();
